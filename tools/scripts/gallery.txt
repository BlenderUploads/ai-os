# Capture the screenshots used in the README.
#
# Each application is shot on its own: close whatever is open, launch one,
# exercise it, photograph it. That is far more reliable than trying to arrange
# cascaded windows by dragging title bars whose positions shift as more
# windows open.

wait 1.5
shot desktop

# --- terminal: ORACLE as a language, and as a persona ---
type (define (square x) (* x x))
wait 0.4
type (map square (range 1 9))
wait 0.4
type (fold + 0 (range 101))
wait 0.4
type (filter (lambda (n) (= 0 (mod n 3))) (range 20))
wait 0.4
type ask what is halcyon
wait 1.4
shot terminal
