# Drive the HALCYON desktop through the QEMU monitor.
#
# Verbs: type <text> | keys <k>... | mon <cmd> | wait <s> | shot <name>
#        moveto <x> <y> | click <x> <y> | drag <x1> <y1> <x2> <y2>
#
# The guest has a relative PS/2 mouse, so moveto/click/drag park the pointer
# at the top-left corner first and walk out from there.

wait 1
shot desktop

# Drag the System Monitor by its title bar, well clear of its buttons.
drag 200 76 600 300
wait 0.5
shot dragged

# Launch both apps from their function keys.
keys f1
wait 0.8
keys f2
wait 0.8
shot launched

# Cycle focus with alt+tab.
keys alt-tab
wait 0.8
shot alt-tab

# Open the launcher menu from the taskbar button.
click 40 753
wait 0.8
shot menu

# Pick the first entry in it.
click 60 700
wait 1.0
shot from-menu
